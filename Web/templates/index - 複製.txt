    <script>
        let globalData = { users: [], orders: [], posts: [], services: [], products: {} };
        let pendingPurchaseItem = null;

        // 開場載入畫面計時器 (設定總共約 7 秒，可自行在總毫秒數調整)
        function initSplashScreen() {
            const splashScreen = document.getElementById('splash-screen');
            const progressBar = document.getElementById('progressBar');
            const progressText = document.getElementById('progressText');
            
            let currentPercent = 0;
            // 7 秒內跑完 100% (總共 7000 毫秒，每 70 毫秒更新一次 = 100 次)
            const totalDuration = 7000; 
            const intervalTime = 70;
            const increment = 100 / (totalDuration / intervalTime);

            let timer = setInterval(() => {
                currentPercent += increment;
                if (currentPercent >= 100) {
                    currentPercent = 100;
                    clearInterval(timer);
                    // 跑完後淡出隱藏
                    splashScreen.style.opacity = '0';
                    setTimeout(() => {
                        splashScreen.style.display = 'none';
                    }, 600); // 對應 CSS 的 transition 時間
                }
                progressBar.style.width = currentPercent + '%';
                progressText.innerText = Math.floor(currentPercent) + '%';
            }, intervalTime);
        }

        async function fetchAllData() {
            try {
                let res = await fetch('/api/data');
                globalData = await res.json();
                if (!globalData.users) globalData.users = [];
                refreshUI();
            } catch (e) {
                console.error("無法同步伺服器資料", e);
            }
        }

        function getCurrentUser() { return localStorage.getItem('scorki_current_user'); }
        function setCurrentUser(name) {
            if (name) localStorage.setItem('scorki_current_user', name);
            else localStorage.removeItem('scorki_current_user');
            updateAuthUI();
        }

        function updateAuthUI() {
            const user = getCurrentUser();
            const userStatus = document.getElementById('userStatus');
            const btnLogin = document.getElementById('btnLogin');
            const btnRegister = document.getElementById('btnRegister');
            const btnLogout = document.getElementById('btnLogout');
            const btnAdmin = document.getElementById('btnAdmin');

            if (user) {
                document.getElementById('welcomeName').innerText = user;
                userStatus.style.display = 'flex';
                btnLogin.style.display = 'none';
                btnRegister.style.display = 'none';
                btnLogout.style.display = 'block';
                btnAdmin.style.display = (user === 'admin') ? 'block' : 'none';
            } else {
                userStatus.style.display = 'none';
                btnLogin.style.display = 'block';
                btnRegister.style.display = 'block';
                btnLogout.style.display = 'none';
                btnAdmin.style.display = 'none';
            }
        }

        function refreshUI() {
            updateAuthUI();
            renderAllProductTables();
            renderBlog();
            renderServices();
        }

        function renderAllProductTables() {
            ['mc-firework', 'mc-other', 'shop-goods', 'other-games'].forEach(cat => {
                const tbody = document.getElementById(cat + '-tbody');
                if (!tbody) return;
                const items = (globalData.products && globalData.products[cat]) || [];
                if (items.length === 0) {
                    tbody.innerHTML = `<tr><td colspan="4" style="text-align:center; color:#9d5c81;">（目前尚無商品）</td></tr>`;
                    return;
                }
                tbody.innerHTML = items.map(item => `
                    <tr><td><b>${item.name}</b></td><td>${item.desc}</td><td>${item.price}</td><td><button class="buy-btn" onclick="buyItem('${cat}', '${item.name}', '${item.price}')">購買</button></td></tr>
                `).join('');
            });
        }

        function renderBlog() {
            const container = document.getElementById('blog-container');
            if (!container) return;
            container.innerHTML = (globalData.posts || []).map(p => `
                <div class="card"><h3>${p.title}</h3><p style="color:var(--text-muted); font-size:12px; margin:5px 0 10px 0;">發布日期：${p.date}</p><p>${p.content.replace(/\n/g, '<br>')}</p></div>
            `).join('');
        }

        function renderServices() {
            const container = document.getElementById('service-container');
            if (!container) return;
            container.innerHTML = (globalData.services || []).map(s => `
                <div class="card" style="display:flex; justify-content:space-between; align-items:center;">
                    <div><h3>🛠️ ${s.name}</h3><p style="color:var(--text-muted); margin-top:5px;">${s.note.replace(/\n/g, '<br>')}</p></div>
                    <a href="/${s.url}" class="btn btn-primary" download>下載檔案</a>
                </div>
            `).join('');
        }

        async function handleRegister(e) {
            e.preventDefault();
            const name = document.getElementById('regName').value.trim();
            const email = document.getElementById('regEmail').value.trim();
            const password = document.getElementById('regPassword').value;

            try {
                let res = await fetch('/api/users', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ name, email, password })
                });
                let result = await res.json();
                
                if (res.ok) {
                    alert('註冊成功！請登入。');
                    closeModal('registerModal');
                    await fetchAllData();
                } else {
                    alert(result.message || '註冊失敗');
                }
            } catch (err) {
                console.error(err);
                alert('連線伺服器失敗');
            }
        }

        async function handleLogin(e) {
            e.preventDefault();
            const name = document.getElementById('loginName').value.trim();
            const password = document.getElementById('loginPassword').value;

            if (name === 'admin' && password === 'admin') {
                alert('管理員登入成功！');
                setCurrentUser('admin');
                closeModal('loginModal');
                return;
            }

            let validUser = (globalData.users || []).find(u => u.name === name && u.password === password);
            
            if (validUser) {
                alert('登入成功！');
                setCurrentUser(validUser.name);
                closeModal('loginModal');
            } else {
                alert('帳號或密碼錯誤！');
            }
        }

        function handleLogout() { setCurrentUser(null); showPage('home'); }

        function buyItem(category, itemName, price) {
            if (!getCurrentUser()) { alert('請先登入！'); openModal('loginModal'); return; }
            if (category.startsWith('mc-')) {
                pendingPurchaseItem = { itemName, price };
                openModal('mcConfirmModal');
            } else {
                sendOrder({ mcId: '(一般會員: ' + getCurrentUser() + ')', itemName, price });
            }
        }

        async function sendOrder(orderData) {
            await fetch('/api/orders', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(orderData)
            });
            alert('訂單已成立並送至後台！');
            fetchAllData();
        }

        async function handleConfirmMcBuy(e) {
            e.preventDefault();
            if (document.getElementById('mcId1').value !== document.getElementById('mcId2').value) {
                alert('兩次 ID 不一致！'); return;
            }
            if (pendingPurchaseItem) {
                await sendOrder({ mcId: document.getElementById('mcId1').value.trim(), itemName: pendingPurchaseItem.itemName, price: pendingPurchaseItem.price });
                closeModal('mcConfirmModal');
                pendingPurchaseItem = null;
            }
        }

        async function renderAdminOrders() {
            let res = await fetch('/api/orders');
            let orders = await res.json();
            document.getElementById('admin-orders-tbody').innerHTML = orders.length === 0 ? `<tr><td colspan="4" style="text-align:center; color:#9d5c81;">（尚無訂單）</td></tr>` :
                orders.map((o, i) => `<tr><td><b>${o.mcId}</b></td><td>${o.itemName}</td><td>${o.price}</td><td><button class="btn btn-danger" style="padding:4px 8px;" onclick="deleteOrder(${i})">🗑️</button></td></tr>`).join('');
        }

        async function deleteOrder(index) {
            await fetch('/api/orders', { method: 'DELETE', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ index }) });
            renderAdminOrders();
        }

        async function renderAdminProductList() {
            let cat = document.getElementById('adminProdCategorySelect').value;
            let res = await fetch(`/api/products/${cat}`);
            let items = await res.json();
            document.getElementById('admin-products-tbody').innerHTML = items.length === 0 ? `<tr><td colspan="4" style="text-align:center; color:#9d5c81;">（尚無商品）</td></tr>` :
                items.map((item, i) => `<tr><td><b>${item.name}</b></td><td>${item.desc}</td><td>${item.price}</td><td><button class="btn btn-danger" style="padding:4px 8px;" onclick="deleteProduct('${cat}', ${i})">🗑️</button></td></tr>`).join('');
        }

        async function deleteProduct(cat, index) {
            await fetch(`/api/products/${cat}`, { method: 'DELETE', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ index }) });
            renderAdminProductList();
            fetchAllData();
        }

        async function handleAddProduct(e) {
            e.preventDefault();
            let cat = document.getElementById('prodLocation').value;
            let item = { name: document.getElementById('prodName').value, desc: document.getElementById('prodDesc').value, price: document.getElementById('prodPrice').value };
            await fetch(`/api/products/${cat}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(item) });
            alert('商品上架成功！');
            fetchAllData();
            renderAdminProductList();
        }

        async function handleAddPost(e) {
            e.preventDefault();
            let post = { title: document.getElementById('postTitle').value, date: new Date().toISOString().split('T')[0], content: document.getElementById('postContent').value };
            await fetch('/api/posts', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(post) });
            alert('文章發布成功！');
            fetchAllData();
        }

        async function handleAddService(e) {
            e.preventDefault();
            let serv = { name: document.getElementById('servName').value, note: document.getElementById('servNote').value, url: document.getElementById('servFileUrl').value };
            await fetch('/api/services', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(serv) });
            alert('服務新增成功！');
            fetchAllData();
        }

        function showPage(pageId) {
            if (pageId === 'admin-panel') {
                if (getCurrentUser() !== 'admin') { alert('無權限'); return; }
                renderAdminOrders();
                renderAdminProductList();
            }
            document.querySelectorAll('.page-content').forEach(p => p.classList.remove('active'));
            document.getElementById(pageId).classList.add('active');
            
            document.querySelectorAll('.nav-links a').forEach(a => a.classList.remove('active'));
            event && event.target && event.target.tagName === 'A' && event.target.classList.add('active');
        }
